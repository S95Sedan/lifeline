cmake_minimum_required(VERSION 3.20)
project(LifelineWindows VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Configure lifeline-windows with a Windows toolchain")
endif()
set(LIFELINE_OPENGL OFF)
set(LIFELINE_PLATFORM "windows")
set(LIFELINE_PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared" "${CMAKE_CURRENT_BINARY_DIR}/shared")
