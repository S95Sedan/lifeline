cmake_minimum_required(VERSION 3.20)
project(LifelineLinux VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Configure lifeline-linux with a Linux toolchain")
endif()
option(LIFELINE_STEAMDECK "Use Steam Deck display and rendering defaults" ON)
option(LIFELINE_OPENGL "Build the native OpenGL renderer" ON)
set(LIFELINE_PLATFORM "linux")
set(LIFELINE_PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shared" "${CMAKE_CURRENT_BINARY_DIR}/shared")
